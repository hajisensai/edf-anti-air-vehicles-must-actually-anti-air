cmake_minimum_required(VERSION 3.20)
project(EDF6AutoTurret VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
add_library(EDF6AutoTurret SHARED src/plugin.cpp src/memory.cpp)
target_include_directories(EDF6AutoTurret PRIVATE third_party/EDFModLoader)
target_compile_options(EDF6AutoTurret PRIVATE /W4 /WX /utf-8 /permissive-)
target_link_options(EDF6AutoTurret PRIVATE /DYNAMICBASE /NXCOMPAT /PDBALTPATH:EDF6AutoTurret.pdb)
set_target_properties(EDF6AutoTurret PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_SOURCE_DIR}/dist/Mods/Plugins")
configure_file(EDF6AutoTurret.ini "${CMAKE_SOURCE_DIR}/dist/Mods/Plugins/EDF6AutoTurret.ini" COPYONLY)
